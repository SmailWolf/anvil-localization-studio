  add_executable(anvil-oodle-helper tools/oodle_helper.cpp)
  set_target_properties(anvil-oodle-helper PROPERTIES AUTOMOC OFF)
  if(MSVC)
    target_compile_options(anvil-oodle-helper PRIVATE /utf-8 /W4)
  else()
    target_link_options(anvil-oodle-helper PRIVATE -municode)
  endif()
  add_dependencies(anvil-extract-breakpoint anvil-oodle-helper)
  add_dependencies(anvil-build-breakpoint-test anvil-oodle-helper)
  install(TARGETS anvil-oodle-helper RUNTIME DESTINATION bin)
else()
  find_program(ANVIL_MINGW_CXX x86_64-w64-mingw32-g++)
  if(ANVIL_MINGW_CXX)
    add_custom_command(OUTPUT "${CMAKE_BINARY_DIR}/anvil-oodle-helper.exe"
      COMMAND "${ANVIL_MINGW_CXX}" -std=c++20 -O2 -municode -static -static-libgcc -static-libstdc++ "${CMAKE_CURRENT_SOURCE_DIR}/tools/oodle_helper.cpp" -o "${CMAKE_BINARY_DIR}/anvil-oodle-helper.exe"
      DEPENDS tools/oodle_helper.cpp VERBATIM)
    add_custom_target(anvil_oodle_helper ALL DEPENDS "${CMAKE_BINARY_DIR}/anvil-oodle-helper.exe")
    add_dependencies(anvil-extract-breakpoint anvil_oodle_helper)
    add_dependencies(anvil-build-breakpoint-test anvil_oodle_helper)
    install(FILES "${CMAKE_BINARY_DIR}/anvil-oodle-helper.exe" DESTINATION bin)
  endif()
